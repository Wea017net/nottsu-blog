{{/* Render hook for Markdown links. - External links (http/https) will get target="_blank" and rel="noopener noreferrer". - Internal links are rendered normally. */}} {{ $href := .Destination | safeURL }} {{ $text := .Text | safeHTML }} {{ if strings.HasPrefix $href "http" }} {{ printf "%s" $href $text | safeHTML -}} {{ else }} {{ printf "%s" $href $text | safeHTML -}} {{ end }}
